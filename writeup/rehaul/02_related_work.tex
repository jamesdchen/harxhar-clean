\item \textbf{Realized variance and the HAR model:}
  $\E_t[\RV_{t+1}] = \beta_0 + \beta_d\, \RV^{(d)}_t + \beta_w\, \RV^{(w)}_t + \beta_m\, \RV^{(m)}_t$, with
  $\RV^{(d)}_t$, $\RV^{(w)}_t$ and $\RV^{(m)}_t$ the average daily realized variance over the last 1, 5 and 22
  trading days. Our HAR components $\bar y^{(k)}_t$ take the same averages of $y$ over $k$ intraday
  intervals.\newline
  \citet{Corsi2009}, \citet{AndersenBollerslevDieboldLabys2003}, \citet{BarndorffNielsenShephard2004},
  \citet{AndersenBollerslevDiebold2007}, \citet{CorsiMittnikPigorschPigorsch2008},

  \citet{HansenHuangShek2012}, \citet{AndersenBollerslevChristoffersenDiebold2006}.

\item \textbf{Intraday periodicity:} volatility follows a U-shaped intraday pattern; the standard adjustment
  divides by a periodicity factor, here $y_{t,i} = \sqrt{\RV_{t,i}/s^2_{t,i}}$.\newline
  \citet{AndersenBollerslev1997} (intraday periodicity), \citet{EngleGallo2006} (intraday measures as indicators of
  daily volatility), \citet{HizmeriBevilacqua2025} (morning volatility uncertainty).

  realized variance, $\E^{\mathbb{Q}}_t[\RV_t] - \E^{\mathbb{P}}_t[\RV_t]$, is positive on average.\newline
  \citet{HizmeriBevilacqua2025}; option pricing and implied volatility \citet{BlackScholes1973},
  \citet{Merton1973}.

\item \textbf{Forecast evaluation:} QLIKE is robust to noise in the volatility proxy: with a conditionally unbiased
  proxy it ranks forecasts as the true conditional variance would, and
  $\argmin_{v > 0} \E_t[\QLIKE(\RV_t, v)] = \E_t[\RV_t]$.\newline
  \citet{Patton2011} (robust loss functions), \citet{DieboldMariano1995} with the small-sample correction of
  \citet{HarveyLeybourneNewbold1997}, \citet{HansenLundeNason2011} (model confidence set), \citet{Ehm2016}
  (Murphy diagrams, elementary scores), \citet{MincerZarnowitz1969} (forecast-efficiency regressions).
  \begin{itemize}
  \item \citet{Hansen2005} (superior predictive ability), \citet{White2000} (reality check), \citet{RomanoWolf2005}
    (StepM), \citet{NeweyWest1987} (HAC variance).

\item \textbf{Shrinkage with many weak signals:}
  \[
    \hat{\bm\beta} = \argmin_{\bm\beta} \lVert \bm y - \bm X\bm\beta \rVert_2^2
      + \lambda\bigl(\alpha \lVert\bm\beta\rVert_1 + \tfrac{1-\alpha}{2} \lVert\bm\beta\rVert_2^2\bigr)
  \]
  (ridge $\alpha = 0$, lasso $\alpha = 1$); multi-penalty ridge gives each block of predictors its own penalty,
  $\lambda \sum_j \omega_j \beta_j^2$ with the penalty weight $\omega_j$ constant within a block.\newline
  \citet{GiannoneLenzaPrimiceri2021} (the illusion of sparsity), \citet{ShenXiu2025} (weak signals),
  \citet{KozakNagelSantosh2020} (shrinking the cross-section), \citet{HoerlKennard1970} (ridge),
  \citet{Tibshirani1996} (lasso), \citet{ZouHastie2005} (elastic net), \citet{vandeWiel2021multipenalty}
  (multi-penalty ridge), \citet{vanWieringen2023ridge}.
  \begin{itemize}
  \item \citet{Zou2006} (adaptive lasso, penalty weights), \citet{FriedmanHastieTibshirani2010} (elastic-net
    parametrization).

\item \textbf{Variable importance:} Shapley additive explanations (SHAP) decompose each forecast additively,
  $\hat y_t = \phi_0 + \sum_{j=1}^{p} \phi_{t,j}$.\newline
  \citet{lundberg2017unified} (SHAP), \citet{lundberg2018consistent} (TreeSHAP), \citet{aas2021explaining}
  (dependent features).
\item \textbf{Retransformation} (one sentence): a smearing estimate would correct $\hat y_t^{\,2} s^2_{t,c}$ for
  retransformation bias; we do not.\newline
  \citet{Duan1983}, \citet{ManningMullahy2001}.
