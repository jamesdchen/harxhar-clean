    \hat{\bm\beta}^{\mathrm{ridge}}_{t,i} &= \argmin_{\beta_0, \bm\beta} \sum_{(t',i') \in \mathcal W_{t,i}}
      (\tilde y_{t',i'} - \beta_0 - \bm x_{t',i'}^\top \bm\beta)^2 + \lambda \sum_{j=1}^{p} \omega_j\, \beta_j^2, \\

\item \textbf{Tree ensembles.} To capture potential nonlinearities and high-order interactions among predictors, we consider two gradient-boosted tree implementations (LightGBM and XGBoost) and random forests.

  \item \textbf{Gradient boosting} \citep{Friedman2001}, implemented via LightGBM \citep{Ke2017} and XGBoost \citep{Chen2016}, minimizes the squared-error loss $\tfrac12 (\tilde y - F)^2$ over boosting rounds $b = 1, \dots, B$ with learning rate $\eta$. Each regression tree $f_b$ partitions the predictor space into $J$ leaves with leaf weights $\bm w$, where leaf $j$ contains the observation subset $\mathcal R_j$ of size $n_j = |\mathcal R_j|$. When the $\ell_1$ leaf penalty $\lambda_1^{\mathrm{tree}} = 0$, the additive update and optimal leaf weight take the form
    \begin{align*}
      F_b &= F_{b-1} + \eta f_b, \\
      w_j^\ast &= -\frac{\sum_{(t',i') \in \mathcal R_j}\bigl(F_{b-1}(\bm x_{t',i'}) - \tilde y_{t',i'}\bigr)}
        {n_j + \lambda_2^{\mathrm{tree}}} .
    \end{align*}
    At round $b$, tree $f_b$ is grown to minimize the regularized second-order objective
    \[
      \sum_{(t',i') \in \mathcal W_{t,i}} \Bigl[\bigl(F_{b-1}(\bm x_{t',i'}) - \tilde y_{t',i'}\bigr)
        f_b(\bm x_{t',i'})
        + \tfrac12 f_b(\bm x_{t',i'})^2\Bigr]
      + \gamma^{\mathrm{tree}} J + \lambda_1^{\mathrm{tree}} \lVert \bm w \rVert_1
      + \tfrac12 \lambda_2^{\mathrm{tree}} \lVert \bm w \rVert_2^2 ,
    \]
    where $\gamma^{\mathrm{tree}}$ penalizes tree complexity per leaf and $\lambda_1^{\mathrm{tree}}, \lambda_2^{\mathrm{tree}}$ apply $\ell_1$ and $\ell_2$ shrinkage to the leaf weights $\bm w$.
  \item Every leaf is required to contain at least $n_{\min}$ observations ($n_j \ge n_{\min}$), which corresponds to the minimum leaf size in LightGBM and, under unit Hessian for squared-error loss, the minimum child weight in XGBoost.
