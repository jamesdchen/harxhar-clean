    $f_b$ has $J$ leaves and leaf weights $\bm w$; leaf $j$ holds the observations $\mathcal R_j$, and
    $n_j = |\mathcal R_j|$. At $\lambda_1^{\mathrm{tree}} = 0$ the fitted function and the leaf weight are
    \begin{align*}
      F_b &= F_{b-1} + \eta f_b, \\
      w_j^\ast &= -\frac{\sum_{t' \in \mathcal R_j}\bigl(F_{b-1}(\bm x_{t'}) - \tilde y_{t'}\bigr)}
        {n_j + \lambda_2^{\mathrm{tree}}} .
    \end{align*}
    The tree minimizes
    \[
      \sum_{t' \in \mathcal W_t} \Bigl[\bigl(F_{b-1}(\bm x_{t'}) - \tilde y_{t'}\bigr) f_b(\bm x_{t'})
        + \tfrac12 f_b(\bm x_{t'})^2\Bigr]
      + \gamma^{\mathrm{tree}} J + \lambda_1^{\mathrm{tree}} \lVert \bm w \rVert_1
      + \tfrac12 \lambda_2^{\mathrm{tree}} \lVert \bm w \rVert_2^2 .
    \]
  \item $\gamma^{\mathrm{tree}}$: penalty per leaf; $\lambda_1^{\mathrm{tree}}, \lambda_2^{\mathrm{tree}}$: L1 and
    L2 penalties on $\bm w$.
  \item Leaf minimum $n_{\min}$: $n_j \ge n_{\min}$. That count is LightGBM's minimum leaf size and, under this
    loss, XGBoost's minimum child weight.
  % 2026-10-15 (consolidation): parked -- N is the minute count N_{t,i} and \mathcal T the out-of-sample day set.

  \item \textbf{Random forest} \citep{Breiman2001}: average of $B$ trees $f_b$, each grown on a bootstrap sample,
    \[
      \hat y_t = \frac1B \sum_{b=1}^{B} f_b(\bm x_t);
