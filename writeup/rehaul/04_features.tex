\item \textbf{Predictors built from each series.}
  \begin{itemize}
  \item Trailing averages of the transformed series $\tilde u$ over horizons of $k$ intervals, lagged one interval
    (HAR components; \citealp{Corsi2009}):
    \[
      \bar u^{(k)}_{t,i} = \frac{1}{k} \sum_{j=1}^{k} \tilde u_{t,i-j}, \qquad k \in \mathcal{K},
    \]
    \[
      \mathcal{K} =
      \begin{cases}
        \{1, 5, 25, 125, 625, 3125\} & \text{closing-interval model},\\
        \{2^0, 2^1, \dots, 2^{11}\} \ \text{(base 2, 12 horizons)} & \text{pooled intraday model}.
      \end{cases}
    \]
  \item $k$ counts intervals of the full intraday grid, overnight included; the closing-interval model uses the
    rows $i = c$.

  \item The same averages of the availability indicator, $\bar a^{(k)}_{t,i}$.
  \item Averages of an activity indicator, $\bar\chi^{(k)}_{t,i}$, for 7 zero-inflated series.
  \item Names in tables and figures: the bar and the interval index are dropped ($i = c$ in the closing-interval
    model), so $u^{(k)}_t$ is series $u$ averaged over $k$ intervals; e.g.\ $\mathrm{VIX}^{(25)}_t$ is the
    transformed VIX averaged over the 25 intervals ending at 15:30 on day $t$. Series symbols: $\mathrm{VIX}$,
    $\mathrm{VIX3M}$, $\mathrm{VVIX}$; constituent aggregates carry EW or VW, as in $\mathrm{OFI}^{\mathrm{EW}}$,

  \end{itemize}

\item \textbf{Transformations, in order:}
  \begin{enumerate}
  \item intraday periodicity adjustment, dividing by a factor estimated on the same intraday interval
    ($\mathcal{D}_{t,i}$ as for $s^2_{t,i}$) and bounded relative to its own trailing median:
    \[
      \tilde u_{t,i} = \frac{u_{t,i}}{b^{(u)}_{t,i}}, \qquad
      b^{(u)}_{t,i} =
      \begin{cases}
        \clip\bigl(\operatorname{mean}_{t' \in \mathcal{D}_{t,i}} u_{t',i},\ 0.5\,\rho_{t,i},\ 2\rho_{t,i}\bigr),
          & u \ge 0,\\[2pt]
        \max\bigl\{\clip\bigl(\operatorname{sd}_{t' \in \mathcal{D}_{t,i}} u_{t',i},\ 0.5\,\rho_{t,i},\
          2\rho_{t,i}\bigr),\ 0.1\,\bar\sigma^{(u)}\bigr\}, & u \text{ signed};
      \end{cases}
    \]
    \begin{itemize}
    \item $\rho_{t,i}$: median of the 250 most recent unbounded factors of interval $i$ (at least 20 required).
    \item $\bar\sigma^{(u)}$: full-sample median of the standard deviation on the same intraday interval.
    \item Factor of a nonnegative series zero or undefined: $b^{(u)}_{t,i} = 1$.
    \item Not adjusted: VIX, VIX3M and VVIX, StockTwits sentiment, FOMC.
    \end{itemize}
    % target.py:186-300. The band acts on exogenous series only (target: slot_band = 0). rho_{t,i} has no extra
    % shift: each baseline in it is already lagged (target.py:201-204). Baselines are estimated on the
    % forward-filled u (ffill executor.py:540 precedes robust_transform :562).
  \item a variance-stabilizing transformation by series type, $\tilde u_{t,i} \leftarrow g(\tilde u_{t,i})$, the
    first matching rule applying:
    \[
      g(v) =
      \begin{cases}
        \sqrt{v} & \text{squared returns, realized variance, turnover,}\\
          & \text{bipower variation, effective spread},\\
        \sign(v)\,|v|^{1/2} & \text{autocovariance},\\
        v^{1/3} & \text{cubed returns},\\
        \sign(v)\,|v|^{1/4} & \text{fourth-power returns},\\
        v & \text{other signed series, absolute returns},\\
        \log \max(v, 10^{-12}) & \text{otherwise}.
      \end{cases}
    \]
    \begin{itemize}
    \item Signed series with no type rule whose standard-deviation divisor sits at its floor on at least 50\% of
      intervals (volatility demand) instead take an inverse hyperbolic sine of the raw series:
      \[
        \tilde u_{t,i} = \operatorname{asinh}(u_{t,i} / \kappa_{t,i}), \qquad
        \kappa_{t,i} = \max\bigl\{ e_{t,i-1},\ 0.01 \operatorname{med}_{j \ge 1} e_{t,i-j} \bigr\}.
      \]
    \item $e_{t,i}$: exponentially weighted moving average of $|u|$ over intervals with $u \ne 0$, half-life
      250 days.
    \end{itemize}
    % target.py:387-442 (map), 303-345 (asinh), 644-663 (routing). asinh replaces both the diurnal division and
    % g: it acts on raw u. The routing share is full-sample. fomc_release in {0,1} maps to {log 1e-12, 0}; after
    % the 1/99 clip adj_fomc_release is constant (outline flag 13).
  \item winsorization at the rolling 1st and 99th percentiles, as for the target:
    \[
      \tilde u_{t,i} \leftarrow \clip\bigl(\tilde u_{t,i},\ Q_{0.01}(\mathcal{U}_{t,i}),\ Q_{0.99}(\mathcal{U}_{t,i})\bigr),
      \qquad \mathcal{U}_{t,i} = \{\tilde u_{t,i-j} : j = 1, \dots, 240\};
    \]
  \item imputation of missing values by the full-sample median, kept alongside the availability indicator:
    \[
      \tilde u_{t,i} \leftarrow
      \begin{cases}
        \tilde u_{t,i}, & a_{t,i} = 1,\\
        \operatorname{med}\{\tilde u_{t',i'} : a_{t',i'} = 1\}, & a_{t,i} = 0;
      \end{cases}
    \]
    % executor.py:602-608.
  \item activity indicators, for the 7 series whose full-sample mode accounts for at least 20\% of intervals:
    \[
      \chi_{t,i} = \mathbf{1}\{u_{t,i} \ne \operatorname{mode}(u)\};
    \]
    % CODE BUG (outline flag 1): u_{t,i} here is the FORWARD-FILLED raw value (executor.py:618-627 runs after the
    % ffill at :540), so chi_{t,i} = 1 on dead tails: fomc_release / fomc_day after 2023-11-01, voldemand after
    % 2023-08-31. Intended: chi_{t,i} = a_{t,i} * 1{u_{t,i} != mode}.
  \item trailing averages $\bar u^{(k)}_{t,i}$, $\bar a^{(k)}_{t,i}$, $\bar\chi^{(k)}_{t,i}$, $\bar y^{(k)}_{t,i}$;
  \item robust standardization on the rolling estimation window (median and interquartile range, with floors),
    for the averages of series values only; written for the closing-interval model, whose rows $t'$ are trading
    days ($i = c$ suppressed), with $\mathcal{W}_t = \{t - W, \dots, t - 1\}$:
    \[
      z^{(k)}_{t} = \frac{\bar u^{(k)}_{t} - \operatorname{med}_{t' \in \mathcal{W}_t} \bar u^{(k)}_{t'}}
        {\max\bigl\{\mathrm{IQR}^{(k)}_{t},\ 0.01 \max_{t' \le t} \mathrm{IQR}^{(k)}_{t'},\
          \mathrm{IQR}^{\ast(k)}_{t}\bigr\}},
      \qquad \mathrm{IQR}^{(k)}_{t} = \mathrm{IQR}\bigl\{\bar u^{(k)}_{t'} : t' \in \mathcal{W}_t\bigr\}.
    \]
    % 2026-10-15 (consolidation): parked -- the window row was s (now t'; s is the periodicity factor):
    % z^{(k)}_{t} = (\bar u^{(k)}_{t} - med_{s \in W_t} \bar u^{(k)}_{s}) / max{...}, IQR^{(k)}_t = IQR{\bar u^{(k)}_s : s \in W_t}.
    \begin{itemize}
    \item $\mathrm{IQR}^{\ast(k)}_{t}$: IQR of $\bar u^{(k)}_{t'}$, $t' < t$, over rows whose $k$-interval averaging
      window is fully observed.
    \item Column whose mode accounts for at least 20\% of rows: $\mathrm{IQR}^{\ast(k)}_{t}$ also excludes rows at

    \item $\bar a^{(k)}$, $\bar\chi^{(k)}$: not standardized. HAR components, calendar dummies:

    \item $t < W$: the first window's statistics.
    \item Pooled intraday model only (rows are intervals $(t', i')$): median and IQR over rows with
      $a_{t',i'} = 1$; fewer than 512 such rows give $z^{(k)} = 0$.
    \end{itemize}
